CREATE DATABASE Managementul_Unei_Banci;

USE Managementul_Unei_Banci;

CREATE TABLE Clienti(
		ClientID int NOT NULL,
        Nume varchar(255) NOT NULL,
        Prenume varchar(255) NOT NULL,
        CNP varchar(255) UNIQUE NOT NULL,
        Email varchar(255),
        Nr_Telefon varchar(255) NOT NULL,
        CONSTRAINT PK_Clienti primary key (ClientID)
);
CREATE TABLE LOCATII(
		LocatieID int NOT NULL,
		Oras varchar(255) NOT NULL,
        Judet varchar(255) NOT NULL,
        CONSTRAINT PK_Locatii primary key (LocatieID)
);
CREATE TABLE Sucursale(
		Locatie int NOT NULL,
		SucursalaID int NOT NULL,
		Nume varchar(255) NOT NULL,
        Adresa varchar(255) NOT NULL,
        Program varchar(255),
        Stat varchar(255),
        CONSTRAINT PK_Sucursala primary key (SucursalaID),
        CONSTRAINT FK_Locatii foreign key (Locatie) references Locatii(LocatieID) on delete cascade
);
CREATE TABLE Angajati(
		AngajatID int NOT NULL,
		Sucursala int NOT NULL,
		Nume varchar(255) NOT NULL,
        Salariul varchar(255) NOT NULL,
        Prenume varchar(255),
        Data_angajare date NOT NULL,
        CONSTRAINT PK_Angajat primary key (AngajatID),
        CONSTRAINT FK_Sucrusala foreign key (Sucursala) references Sucursale(SucursalaID) on delete cascade
);
CREATE TABLE Conturi(
		ContID int NOT NULL,
		Clienti int NOT NULL,
        Sucursala int,
		Iban varchar(255) NOT NULL UNIQUE,
        Moneda varchar(255) NOT NULL,
        Stat varchar(255) not null,
        Tip bool not null,
        CONSTRAINT PK_Cont primary key (ContID),
        CONSTRAINT FK_Clienti foreign key (Clienti) references Clienti(ClientID) on delete cascade,
        CONSTRAINT FK_Sucrusala2 foreign key (Sucursala) references Sucursale(SucursalaID) on delete set NULL
);
CREATE TABLE Depozite(
		DepozitID int NOT NULL,
		Cont int NOT NULL,
		Data_Incepere timestamp NOT NULL,
		Data_Terminare timestamp NOT NULL,
        Penalizare float,
        Dobanda float not null,
        CONSTRAINT PK_Depozit primary key (DepozitID),
        CONSTRAINT FK_ContDepozit foreign key (Cont) references Conturi(Contid) on delete cascade
);
CREATE TABLE Tranzactii(
		TranzactieID int NOT NULL,
		Cont int NOT NULL,
		Nume varchar(255) not null,
		Data_tranz timestamp not null,
        Suma float not null,
        Iban_participant varchar(255),
        Stat varchar(255),
        CONSTRAINT PK_Tranzactie primary key (TranzactieID),
        CONSTRAINT FK_ContTranzactie foreign key (Cont) references Conturi(Contid) on delete cascade
);
CREATE TABLE Carduri(
		CardID int NOT NULL,
		Cont int NOT NULL,
		Data_Expirare date Not NULL,
        Cod_Securitate varchar(255) NOT NULL,
        Tip_Card bool not null,
        Dobanda float,
        CONSTRAINT PK_Depozit primary key (CardID),
        CONSTRAINT FK_ContCard foreign key (Cont) references Conturi(Contid) on delete cascade
);
